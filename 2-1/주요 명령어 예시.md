# 1. 거래 추가 (대화형)
python -m budget_app add

# 2. 거래 목록 (스트리밍, 최신순)
python -m budget_app list --limit 5

# 3. 거래 검색
python -m budget_app search --from 2026-08-01 --to 2026-08-31 --category 식비

# 4. 월별 요약 및 통계
python -m budget_app summary --month 2026-08 --top 3

# 5. 예산 설정
python -m budget_app budget set --month 2026-08 --amount 500000

# 6. 카테고리 관리
python -m budget_app category list
python -m budget_app category add 취미
python -m budget_app category remove 식비 --fallback 기타

# 7. 거래 수정 / 삭제
python -m budget_app update --id <거래ID>
python -m budget_app delete --id <거래ID>

# 8. 가져오기 / 내보내기
python -m budget_app export --out backup.csv --month 2026-08
python -m budget_app import --from backup.csv
